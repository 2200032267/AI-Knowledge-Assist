<script>
  const homepage = document.getElementById('homepage');
  const authModal = document.getElementById('authModal');
  const closeModal = document.getElementById('closeModal');
  const tabs = document.querySelectorAll('.tab');
  const signinForm = document.getElementById('signinForm');
  const signupForm = document.getElementById('signupForm');
  const navLoginBtn = document.getElementById('navLoginBtn');
  const heroCta = document.getElementById('heroCta');
  const toast = document.getElementById('toast');
  const toastMessage = document.getElementById('toastMessage');

  let pendingAction = null;

  // Modal Controls
  function openModal(callback = null) {
    pendingAction = callback;
    authModal.classList.add('active');
    homepage.classList.add('blurred');
    document.body.style.overflow = 'hidden';
  }

  function closeModalHandler() {
    authModal.classList.remove('active');
    homepage.classList.remove('blurred');
    document.body.style.overflow = '';
    resetForms();
  }

  // Handle protected actions
  function requireAuth(action) {
    const isLoggedIn = localStorage.getItem('isLoggedIn') === 'true';
    if (!isLoggedIn) {
      openModal(action);
      return false;
    }
    action();
    return true;
  }

  // Tab Switching
  tabs.forEach(tab => {
    tab.addEventListener('click', () => {
      const target = tab.dataset.tab;
      
      tabs.forEach(t => t.classList.remove('active'));
      tab.classList.add('active');
      
      document.querySelectorAll('.form-container').forEach(form => {
        form.classList.remove('active');
      });
      
      document.getElementById(target + 'Form').classList.add('active');
      resetErrors();
    });
  });

  // Form Validation
  function validateEmail(email) {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
  }

  function showError(inputId, errorId, show = true) {
    const input = document.getElementById(inputId);
    const error = document.getElementById(errorId);
    if (show) {
      input.classList.add('error');
      error.classList.add('show');
    } else {
      input.classList.remove('error');
      error.classList.remove('show');
    }
  }

  function resetErrors() {
    document.querySelectorAll('.form-input').forEach(input => {
      input.classList.remove('error');
    });
    document.querySelectorAll('.error-text').forEach(error => {
      error.classList.remove('show');
    });
  }

  function resetForms() {
    signinForm.reset();
    signupForm.reset();
    resetErrors();
  }

  // Toast
  function showToast(message) {
    toastMessage.textContent = message;
    toast.classList.add('show');
    setTimeout(() => {
      toast.classList.remove('show');
    }, 3000);
  }

  // Handle Auth Success
  function handleAuthSuccess(message) {
    localStorage.setItem('isLoggedIn', 'true');
    closeModalHandler();
    showToast(message);

      setTimeout(() => {
        pendingAction();
        pendingAction = null;
      }, 300);
    }

    let isValid = true;
    
    if (!validateEmail(email)) {
      showError('signinEmail', 'signinEmailError');
      isValid = false;
    }
    
    if (password.length < 6) {
      showError('signinPassword', 'signinPasswordError');
      isValid = false;
    }

      // Simulate API call
      setTimeout(() => {
        handleAuthSuccess('Welcome back!');

    const email = document.getElementById('signupEmail').value.trim();
    const password = document.getElementById('signupPassword').value;
    const confirm = document.getElementById('signupConfirm').value;
    
    let isValid = true;
    
    if (name.length < 2) {
      showError('signupName', 'signupNameError');
      isValid = false;
    }
    
    if (!validateEmail(email)) {
      showError('signupEmail', 'signupEmailError');
      isValid = false;
    }
    
    if (password.length < 6) {
      showError('signupPassword', 'signupPasswordError');
      isValid = false;
    }
    
    if (password !== confirm) {
      showError('signupConfirm', 'signupConfirmError');
      isValid = false;
    }
    
    if (isValid) {
      // Simulate API call
      setTimeout(() => {
        handleAuthSuccess('Account created successfully!');
      }, 500);
    }
  });
