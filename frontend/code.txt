import React, { useState } from 'react';
import { Mail, Lock, User, Eye, EyeOff, X } from 'lucide-react';
import toast from 'react-hot-toast';

const AuthModal = ({ onClose, onLogin }) => {
  const [isSignUp, setIsSignUp] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [rememberMe, setRememberMe] = useState(false);
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    password: ''
  });

  // Get all users from localStorage
  const getUsers = () => JSON.parse(localStorage.getItem('users') || '[]');
  
  const saveUsers = (users) => localStorage.setItem('users', JSON.stringify(users));

  const validateEmail = (email) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);

  const handleSubmit = (e) => {
    e.preventDefault();
    const { name, email, password } = formData;

    // 1. VALIDATION
    if (!validateEmail(email)) {
      toast.error('Please enter a valid email', {
        style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
      });
      return;
    }

    if (password.length < 6) {
      toast.error('Password must be at least 6 characters', {
        style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
      });
      return;
    }

    const users = getUsers();

    // 2. SIGN UP LOGIC
    if (isSignUp) {
      if (!name.trim()) {
        toast.error('Please enter your name', {
          style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
        });
        return;
      }

      // Check if email already exists
      if (users.find(u => u.email.toLowerCase() === email.toLowerCase())) {
        toast.error('Email already exists. Try signing in instead', {
          style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
        });
        return;
      }

      // Create new user
      const newUser = {
        id: Date.now().toString(),
        name: name.trim(),
        email: email.toLowerCase(),
        password, // In prod: hash this. For localStorage demo it's ok
        createdAt: new Date().toISOString()
      };

      users.push(newUser);
      saveUsers(users);
      
      // Auto login after signup
      loginUser(newUser);
      toast.success(`Welcome, ${newUser.name}!`, {
        style: { background: '#1ABC9C', color: '#000' }
      });
      return;
    }

    // 3. SIGN IN LOGIC
    const user = users.find(u => u.email.toLowerCase() === email.toLowerCase());
    
    if (!user) {
      toast.error('No account found with this email', {
        style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
      });
      return;
    }

    if (user.password !== password) {
      toast.error('Incorrect password. Please try again', {
        style: { background: '#1A1A1A', color: '#fff', border: '1px solid #2A2A2A' }
      });
      return;
    }

    // Login success
    loginUser(user);
    toast.success(`Welcome back, ${user.name}!`, {
      style: { background: '#1ABC9C', color: '#000' }
    });
  };

  // REMEMBER ME: localStorage = persistent, sessionStorage = tab only
  const loginUser = (user) => {
    if (rememberMe) {
      localStorage.setItem('session', user.id); // Persists after browser close
      localStorage.setItem('rememberMe', 'true');
    } else {
      sessionStorage.setItem('session', user.id); // Clears on tab close
      localStorage.removeItem('rememberMe');
    }
    onLogin(user);
    onClose();
  };

  // Auto-fill email if remembered
  React.useEffect(() => {
    const remembered = localStorage.getItem('rememberMe');
    if (remembered) setRememberMe(true);
  }, []);

  return (
    <div className="fixed inset-0 bg-black/80 backdrop-blur-sm flex items-center justify-center z-50 p-4">
      <div className="bg-[#0D0D0D] border border-[#2A2A2A] rounded-2xl w-full max-w-md p-8 relative">
        <button 
          onClick={onClose}
          className="absolute top-4 right-4 text-[#B0B0B0] hover:text-white transition"
        >
          <X size={20} />
        </button>

        <div className="text-center mb-8">
          <div className="w-12 h-12 bg-[#1ABC9C] rounded-xl flex items-center justify-center mx-auto mb-4">
            <svg className="w-6 h-6 text-black" fill="currentColor" viewBox="0 0 20 20">
              <path d="M3 4a1 1 0 011-1h12a1 1 0 011 1v2a1 1 0 01-1 1H4a1 1 0 01-1-1V4zM3 10a1 1 0 011-1h6a1 1 0 011 1v6a1 1 0 01-1 1H4a1 1 0 01-1-1v-6zM14 9a1 1 0 00-1 1v6a1 1 0 001 1h2a1 1 0 001-1v-6a1 1 0 00-1-1h-2z" />
            </svg>
          </div>
          <h2 className="text-2xl font-bold text-white">Welcome to AI Knowledge Assistant</h2>
        </div>

        {/* Tabs */}
        <div className="flex border-b border-[#2A2A2A] mb-6">
          <button
            onClick={() => setIsSignUp(false)}
            className={`flex-1 py-3 text-sm font-medium transition ${
              !isSignUp 
              ? 'text-[#1ABC9C] border-b-2 border-[#1ABC9C]' 
                : 'text-[#B0B0B0] hover:text-white'
            }`}
          >
            Sign In
          </button>
          <button
            onClick={() => setIsSignUp(true)}
            className={`flex-1 py-3 text-sm font-medium transition ${
              isSignUp 
              ? 'text-[#1ABC9C] border-b-2 border-[#1ABC9C]' 
                : 'text-[#B0B0B0] hover:text-white'
            }`}
          >
            Create Account
          </button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          {isSignUp && (
            <div>
              <label className="block text-sm text-[#B0B0B0] mb-2">Full Name</label>
              <div className="relative">
                <User className="absolute left-3 top-1/2 -translate-y-1/2 text-[#B0B0B0]" size={18} />
                <input
                  type="text"
                  value={formData.name}
                  onChange={(e) => setFormData({...formData, name: e.target.value})}
                  placeholder="John Doe"
                  className="w-full bg-[#1A1A1A] border border-[#2A2A2A] text-white rounded-lg pl-10 pr-4 py-3 
                           focus:border-[#1ABC9C] focus:ring-1 focus:ring-[#1ABC9C] outline-none"
                />
              </div>
            </div>
          )}

          <div>
            <label className="block text-sm text-[#B0B0B0] mb-2">Email</label>
            <div className="relative">
              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 text-[#B0B0B0]" size={18} />
              <input
                type="email"
                value={formData.email}
                onChange={(e) => setFormData({...formData, email: e.target.value})}
                placeholder="you@example.com"
                className="w-full bg-[#1A1A1A] border border-[#2A2A2A] text-white rounded-lg pl-10 pr-4 py-3 
                         focus:border-[#1ABC9C] focus:ring-1 focus:ring-[#1ABC9C] outline-none"
              />
            </div>
          </div>

          <div>
            <label className="block text-sm text-[#B0B0B0] mb-2">Password</label>
            <div className="relative">
              <Lock className="absolute left-3 top-1/2 -translate-y-1/2 text-[#B0B0B0]" size={18} />
              <input
                type={showPassword? 'text' : 'password'}
                value={formData.password}
                onChange={(e) => setFormData({...formData, password: e.target.value})}
                placeholder="••••"
                className="w-full bg-[#1A1A1A] border border-[#2A2A2A] text-white rounded-lg pl-10 pr-10 py-3 
                         focus:border-[#1ABC9C] focus:ring-1 focus:ring-[#1ABC9C] outline-none"
              />
              <button
                type="button"
                onClick={() => setShowPassword(!showPassword)}
                className="absolute right-3 top-1/2 -translate-y-1/2 text-[#B0B0B0] hover:text-white"
              >
                {showPassword? <EyeOff size={18} /> : <Eye size={18} />}
              </button>
            </div>
          </div>

          {!isSignUp && (
            <div className="flex items-center justify-between">
              <label className="flex items-center gap-2 cursor-pointer">
                <input
                  type="checkbox"
                  checked={rememberMe}
                  onChange={(e) => setRememberMe(e.target.checked)}
                  className="w-4 h-4 accent-[#1ABC9C] rounded bg-[#1A1A1A] border-[#2A2A2A]"
                />
                <span className="text-sm text-[#B0B0B0]">Remember me</span>
              </label>
              {/* REMOVED: Forgot password - can't work without backend */}
            </div>
          )}

          <button
            type="submit"
            className="w-full bg-[#1ABC9C] text-black font-semibold py-3 rounded-lg hover:bg-[#16A085] transition"
          >
            {isSignUp? 'Create Account' : 'Sign In'}
          </button>
        </form>

        {/* REMOVED: Google/GitHub buttons - needs OAuth backend */}
        
        <p className="text-xs text-[#666] text-center mt-6">
          By continuing you agree to our <span className="text-[#1ABC9C]">Terms</span> and <span className="text-[#1ABC9C]">Privacy Policy</span>
        </p>
      </div>
    </div>
  );
};

export default AuthModal;
