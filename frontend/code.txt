import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { 
  Menu, X, Layers, Code, FileText, Scissors, Circle, Database, Search, 
  Cpu, MessageSquare, ArrowRight, Github, ChevronRight, Book, Zap,

    const handleScroll = () => {
      const sections = ['docs', 'features', 'architecture', 'stack', 'how-it-works'];
      const scrollY = window.scrollY + 100;
      
      for (const section of sections) {
        const el = document.getElementById(section);
        if (el && scrollY >= el.offsetTop && scrollY < el.offsetTop + el.offsetHeight) {
          setActiveSection(section);
          break;
        }
      }
    };
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry, idx) => {
          if (entry.isIntersecting) {
            setTimeout(() => entry.target.classList.add('visible'), idx * 100);
          }
        });
      },
      { threshold: 0.1 }
    );

    document.querySelectorAll('.fade-in').forEach(el => observer.observe(el));
    return () => observer.disconnect();
  }, []);

    setMobileMenuOpen(false);
    const el = document.getElementById(id);
    if (el) {
      const offset = 80;
      window.scrollTo({ top: el.offsetTop - offset, behavior: 'smooth' });
    }

    { id: 'docs', label: 'Docs' },

      {/* 1. DOCS SECTION */}

                    <li key={i} className="flex items-start gap-3 text-sm text-[#B0B0B0]">

              <React.Fragment key={idx}>
