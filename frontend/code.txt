import React, { useState, useEffect, useRef } from 'react';
import { Mic, MicOff, Loader2 } from 'lucide-react';
import toast from 'react-hot-toast';

const VoiceInput = ({ 
  onTranscript, // Updates input box text
  onAutoSend, // Fires when user stops talking
  disabled = false // Disable during AI response
}) => {
  const [isListening, setIsListening] = useState(false);
  const [isSupported, setIsSupported] = useState(true);
  const recognitionRef = useRef(null);
  const silenceTimerRef = useRef(null);
  const finalTranscriptRef = useRef('');

  useEffect(() => {
    // Check browser support
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    
    if (!SpeechRecognition) {
      setIsSupported(false);
      console.warn('Speech recognition not supported in this browser');
      return;
    }

    // Init recognition
    recognitionRef.current = new SpeechRecognition();
    recognitionRef.current.continuous = true; // Keep listening
    recognitionRef.current.interimResults = true; // Real-time results
    recognitionRef.current.lang = 'en-US';
    recognitionRef.current.maxAlternatives = 1;

    // When speech is detected
    recognitionRef.current.onresult = (event) => {
      let interimTranscript = '';
      let finalTranscript = '';

      for (let i = event.resultIndex; i < event.results.length; i++) {
        const transcript = event.results[i][0].transcript;
        if (event.results[i].isFinal) {
          finalTranscript += transcript + ' ';
        } else {
          interimTranscript += transcript;
        }
      }

      // Update input box with current transcript
      const fullText = finalTranscriptRef.current + finalTranscript + interimTranscript;
      onTranscript(fullText);

      // Reset silence timer - user is still talking
      if (silenceTimerRef.current) {
        clearTimeout(silenceTimerRef.current);
      }

      // If we have final text, start silence detection
      if (finalTranscript) {
        finalTranscriptRef.current += finalTranscript;
        
        // Auto-send after 1.5s of silence
        silenceTimerRef.current = setTimeout(() => {
          if (finalTranscriptRef.current.trim()) {
            onAutoSend(finalTranscriptRef.current.trim());
            finalTranscriptRef.current = '';
            stopListening();
          }
        }, 1500);
      }
    };

    // Handle errors
    recognitionRef.current.onerror = (event) => {
      console.error('Speech error:', event.error);
      if (event.error === 'not-allowed') {
        toast.error('Mic permission denied. Enable in browser settings.');
      } else if (event.error === 'no-speech') {
        toast.error('No speech detected. Try again.');
      } else {
        toast.error('Voice input error. Try again.');
      }
      stopListening();
    };

    // When recognition ends
    recognitionRef.current.onend = () => {
      setIsListening(false);
      if (silenceTimerRef.current) {
        clearTimeout(silenceTimerRef.current);
      }
    };

    return () => {
      if (recognitionRef.current) {
        recognitionRef.current.stop();
      }
      if (silenceTimerRef.current) {
        clearTimeout(silenceTimerRef.current);
      }
    };
  }, [onTranscript, onAutoSend]);

  const startListening = async () => {
    if (!isSupported) {
      toast.error('Voice input not supported in Firefox. Use Chrome/Edge.');
      return;
    }

    try {
      // Request mic permission
      await navigator.mediaDevices.getUserMedia({ audio: true });
      
      finalTranscriptRef.current = '';
      onTranscript(''); // Clear input
      recognitionRef.current?.start();
      setIsListening(true);
      toast.success('Listening... Speak now', { duration: 2000 });
    } catch (err) {
      toast.error('Mic access denied');
    }
  };

  const stopListening = () => {
    recognitionRef.current?.stop();
    setIsListening(false);
    if (silenceTimerRef.current) {
      clearTimeout(silenceTimerRef.current);
    }
  };

  const toggleListening = () => {
    if (isListening) {
      stopListening();
    } else {
      startListening();
    }
  };

  if (!isSupported) {
    return (
      <button
        disabled
        className="p-2 text-[#444] cursor-not-allowed"
        title="Voice not supported in Firefox"
      >
        <MicOff size={20} />
      </button>
    );
  }

  return (
    <button
      onClick={toggleListening}
      disabled={disabled}
      className={`p-2 rounded-lg transition-all duration-200 ${
        isListening 
        ? 'bg-red-500/20 text-red-400 animate-pulse' 
          : 'text-[#B0B0B0] hover:bg-[#2A2A2A] hover:text-white'
      } disabled:opacity-50 disabled:cursor-not-allowed`}
      title={isListening? 'Stop recording' : 'Voice input'}
    >
      {isListening? <MicOff size={20} /> : <Mic size={20} />}
    </button>
  );
};

export default VoiceInput;

const ChatInput = () => {
  const [input, setInput] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const { handleSendMessage } = useChat();

  // Fired when voice stops for 1.5s
  const handleVoiceAutoSend = (transcript) => {
    if (transcript.trim()) {
      handleSendMessage(transcript);
      setInput(''); // Clear after send
    }
  };

  const handleSend = () => {
    if (!input.trim() || isLoading) return;
    handleSendMessage(input);
    setInput('');
  };

  const handleKeyDown = (e) => {
    if (e.key === 'Enter' &&!e.shiftKey) {
      e.preventDefault();
      handleSend();
    }
  };

  return (
    <div className="border-t border-[#1A1A1A] p-4">
      <div className="max-w-3xl mx-auto">
        <div className="flex items-end gap-2 bg-[#1A1A1A] rounded-lg p-2 border border-[#2A2A2A]">
          <button className="p-2 text-[#B0B0B0] hover:text-white">
            <Upload size={20} />
          </button>
          
          <VoiceInput 
            onTranscript={setInput} // Updates input as you speak
            onAutoSend={handleVoiceAutoSend} // Sends when you stop talking
            disabled={isLoading}
          />

          <textarea
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={handleKeyDown}
            placeholder="Ask me anything..."
            rows={1}
            className="flex-1 bg-transparent text-white outline-none resize-none max-h-32 py-2"
            disabled={isLoading}
          />
          
          <button
            onClick={handleSend}
            disabled={!input.trim() || isLoading}
            className="p-2 bg-[#1ABC9C] text-black rounded-lg hover:bg-[#16A085] disabled:opacity-50 disabled:cursor-not-allowed"
          >
            <Send size={20} />
          </button>
        </div>
        <p className="text-xs text-[#666] text-center mt-2">
          AI can make mistakes. Verify important info.
        </p>
      </div>
    </div>
  );
};
